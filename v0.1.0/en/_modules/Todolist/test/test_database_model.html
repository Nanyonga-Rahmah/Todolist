 18def test_database_model_gets_all_todos(setup) -> None:
 19    """Test all todos are retrieved."""
 20    with setup.app_context():
 21        todo1 = Todo(
 22            title=fake.sentence(nb_words=3),
 23            description=fake.paragraph(nb_sentences=2),
 24            completed=fake.boolean(),
 25        )
 26        todo2 = Todo(
 27            title=fake.sentence(nb_words=3),
 28            description=fake.paragraph(nb_sentences=2),
 29            completed=fake.boolean(),
 30        )
 31        db.session.add(todo1)
 32        db.session.add(todo2)
 33        db.session.commit()
 34
 35        todos = db.session.execute(db.select(Todo)).scalars().all()
 36        assert len(todos) == 2

 39def test_database_model_add_todo(setup) -> None:
 40    """Test a todo is added to the database."""
 41    with setup.app_context():
 42        todo = Todo(
 43            title=fake.sentence(nb_words=3),

 47        db.session.add(todo)
 48        db.session.commit()
 49
